\begin{abstract}
Stochastic differential equations (SDEs) provide a natural framework for modeling environmental and biological processes subject to intrinsic variability, turbulent dispersion, and random external perturbations. In this report, we formulate and investigate continuous stochastic systems describing biological waste degradation, hydrological streamflow/pollutant transport, and indoor air contaminant dynamics. We construct It\^o diffusion formulations, identify physical interpretations for drift and diffusion operators, and analyze numerical simulations performed via the Euler--Maruyama integration scheme.
\end{abstract}

\section{Motivation and Background}
\begin{itemize}
    \item \textbf{Physical System:} Real-world degradation kinetics, open-channel streamflow pollutant mixing, and indoor particulate ventilation operate under continuous environmental fluctuations and irregular driving loads.
    \item \textbf{Why Stochastic Modeling?} Deterministic reaction-diffusion equations assume homogenous reaction rates and smooth flow velocities. Incorporating Brownian motion accounts for microscopic uncertainties, thermal turbulence, and non-uniform boundary influxes.
\end{itemize}

Foundational mathematical treatments of environmental SDE systems trace back to stochastic degradation models and multi-compartment diffusion dynamics \cite{kloeden1992}. 

\section{Model Description}
\begin{definition}
Let $X_t$ denote the pollutant or substrate concentration at time $t \ge 0$. The governing It\^o stochastic differential equation is expressed as:
\begin{equation}
\mathrm{d}X_t = \mu(X_t, t)\,\mathrm{d}t + \sigma(X_t, t)\,\mathrm{d}W_t, \quad X_0 = x_0
\label{eq:sde}
\end{equation}
where $\mu(X_t, t)$ represents the deterministic reaction drift, $\sigma(X_t, t)$ denotes the state-dependent diffusion coefficient, and $W_t$ is a standard one-dimensional Wiener process satisfying:
\begin{align}
\mathbb{E}[\mathrm{d}W_t] &= 0, \\
\mathbb{E}[(\mathrm{d}W_t)^2] &= \mathrm{d}t.
\end{align}
\end{definition}

\subsection{Numerical Integration Scheme}
\begin{theorem}
Given a temporal discretization $0 = t_0 < t_1 < \dots < t_N = T$ with uniform step size $\Delta t = t_{n+1} - t_n$, the Euler--Maruyama discrete approximation follows the recurrence:
\begin{equation}
X_{n+1} = X_n + \mu(X_n, t_n)\Delta t + \sigma(X_n, t_n)\Delta W_n
\end{equation}
where the increments $\Delta W_n = W_{t_{n+1}} - W_{t_n} \sim \mathcal{N}(0, \Delta t)$ are independent and identically distributed.
\end{theorem}
